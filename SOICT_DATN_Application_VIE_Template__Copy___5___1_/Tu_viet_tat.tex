\chapter*{DANH MỤC CÁC TỪ VIẾT TẮT}
\begin{itemize}
    \item \textbf{API} -- Application Programming Interface (Giao diện lập trình ứng dụng)
    \item \textbf{REST API} -- Representational State Transfer API
    \item \textbf{JWT} -- JSON Web Token (Mã thông báo web JSON)
    \item \textbf{RBAC} -- Role-Based Access Control (Kiểm soát truy cập dựa trên vai trò)
    \item \textbf{NFC} -- Near Field Communication (Giao tiếp trường gần)
    \item \textbf{RFID} -- Radio Frequency Identification (Xác định bằng tần số vô tuyến)
    \item \textbf{CCCD} -- Căn Cước Công Dân
    \item \textbf{ICAO} -- International Civil Aviation Organization (Tổ chức hàng không dân dụng quốc tế)
    \item \textbf{MQTT} -- Message Queuing Telemetry Transport (Giao thức truyền tải điều độc hàng đợi tin nhắn)
    \item \textbf{BAC} -- Basic Access Control (Kiểm soát truy cập cơ bản)
    \item \textbf{PACE} -- Password Authenticated Connection Establishment
    \item \textbf{DG2} -- Data Group 2 (Nhóm dữ liệu 2 - ảnh chân dung CCCD)
    \item \textbf{MRZ} -- Machine Readable Zone (Vùng đọc được bằng máy)
    \item \textbf{LDS} -- Logical Data Structure (Cấu trúc dữ liệu logic)
    \item \textbf{SOD} -- Security Object Data (Dữ liệu đối tượng bảo mật)
    \item \textbf{ArcFace} -- Additive Angular Margin Face Loss (Hàm mất má góc phạt cộng cho nhận diện khuôn mặt)
    \item \textbf{FaceNet} -- Mô hình nhận diện khuôn mặt sử dụng Triplet Loss
    \item \textbf{DCNN} -- Deep Convolutional Neural Network (Mạng nơ-ron tích chập sâu)
    \item \textbf{PCA} -- Principal Component Analysis (Phân tích thành phần chính)
    \item \textbf{LBPH} -- Local Binary Patterns Histograms (Biểu đồ mẫu nhị phân cục bộ)
    \item \textbf{IoT} -- Internet of Things (Internet vạn vật)
    \item \textbf{ESP32} -- Vi điều khiển hỗ trợ Wi-Fi và Bluetooth của hãng Espressif
    \item \textbf{WebSocket} -- Giao thức truyền thông hai chiều thời gian thực
    \item \textbf{STOMP} -- Simple Text Oriented Messaging Protocol (Giao thức nhắn tin định hướng văn bản đơn giản)
    \item \textbf{Spring Boot} -- Framework phát triển ứng dụng Java
    \item \textbf{React} -- Thư viện JavaScript để xây dựng giao diện người dùng
    \item \textbf{Flutter} -- Framework phát triển ứng dụng di động đa nền tảng
    \item \textbf{Dart} -- Ngôn ngữ lập trình được sử dụng trong Flutter
    \item \textbf{Python Flask} -- Framework web nhẹ cho Python
    \item \textbf{InsightFace} -- Thư viện nhận diện khuôn mặt chuyên sâu
    \item \textbf{TensorFlow Lite} -- Phiên bản tối ưu hóa của TensorFlow cho thiết bị di động
    \item \textbf{JSON} -- JavaScript Object Notation (Ký pháp đối tượng JavaScript)
    \item \textbf{HTTP} -- HyperText Transfer Protocol (Giao thức truyền tải siêu văn bản)
    \item \textbf{HTTPS} -- HTTP Secure (HTTP an toàn)
    \item \textbf{SSL/TLS} -- Secure Sockets Layer / Transport Layer Security (Lớp ổ cắm an toàn / Bảo mật lớp vận chuyển)
    \item \textbf{MySQL} -- Hệ quản trị cơ sở dữ liệu quan hệ mã nguồn mở
    \item \textbf{SQL} -- Structured Query Language (Ngôn ngữ truy vấn có cấu trúc)
    \item \textbf{CRUD} -- Create, Read, Update, Delete (Tạo, đọc, cập nhật, xóa)
    \item \textbf{UML} -- Unified Modeling Language (Ngôn ngữ mô hình hóa thống nhất)
    \item \textbf{OOP} -- Object-Oriented Programming (Lập trình hướng đối tượng)
    \item \textbf{MVC} -- Model-View-Controller (Mô hình-Khung nhìn-Bộ điều khiển)
    \item \textbf{SOA} -- Service-Oriented Architecture (Kiến trúc hướng dịch vụ)
    \item \textbf{MVP} -- Minimum Viable Product (Sản phẩm khả dụng tối thiểu)
    \item \textbf{QoS} -- Quality of Service (Chất lượng dịch vụ)
    \item \textbf{LFW} -- Labeled Faces in the Wild (Khuôn mặt được gắn nhãn trong tự nhiên - tập dữ liệu)
    \item \textbf{Cosine Similarity} -- Độ tương đồng Cosine (độ đo mức độ tương tự giữa hai vector)
\end{itemize}
